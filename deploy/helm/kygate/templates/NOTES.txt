  kubectl logs -n {{ .Release.Namespace }} deploy/{{ include "kygate.fullname" . }}-control | grep -E 'Virtual key|Admin token'

  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "kygate.fullname" . }}-gateway 4000:{{ .Values.gateway.service.port }}

  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "kygate.fullname" . }}-control 4001:{{ .Values.control.service.port }}
